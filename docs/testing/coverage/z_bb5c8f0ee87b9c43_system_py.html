1""" 

2Ресурс «Система» (System): публічна конфігурація для фронтенду (GET /api/v1/config). 

3 

4Без автентифікації. Ліміти тут мають збігатися з контрактом (docs/api/contract.md, §2.2 і §3.0),

8import os 

9from typing import Dict, Optional 

10 

11from fastapi import APIRouter, Response 

12from pydantic import BaseModel, Field 

13 

14router = APIRouter(prefix="/api/v1", tags=["System"]) 

15 

16API_VERSION = "1.0.0" 

17 

18 

19class Limits(BaseModel): 

20 total_credits: int = Field(240, description="Кредитів за всю програму (4 роки)") 

21 free_choice_credits: int = Field(25, description="Ліміт вільного вибору за всі роки") 

22 professional_credits: int = Field(54, description="Орієнтир профорієнтовних кредитів") 

23 min_year_credits: int = Field(57, description="Мінімум кредитів на рік") 

24 max_directions: int = Field(3, description="Максимум напрямків розвитку") 

25 max_plan_items: int = Field(30, description="Максимум дисциплін у плані") 

26 max_item_credits: int = Field(30, description="Максимум кредитів однієї дисципліни") 

27 academic_year_min: int = Field(2024, description="Мінімальний навчальний рік") 

28 academic_year_max: int = Field(2030, description="Максимальний навчальний рік") 

29 max_upload_bytes: int = Field(10 * 1024 * 1024, description="Максимальний розмір документа деканату") 

30 

31 

32class LoadLevelHours(BaseModel): 

33 min_hours_per_week: Optional[int] = Field(None, description="Нижня межа годин на тиждень") 

34 max_hours_per_week: int = Field(..., description="Верхня межа годин на тиждень")

37class Features(BaseModel): 

38 ai_generation: bool = Field(False, description="AI-генерація плану (етап 2)") 

39 certificate_programs: bool = Field(False, description="Сертифікатні програми (етап 2)") 

40 saz_sync: bool = Field(False, description="Синхронізація з САЗ (заблоковано, R14)")

44class ConfigResponse(BaseModel): 

45 api_version: str = Field(API_VERSION, description="Версія API") 

46 google_client_id: str = Field(..., description="Публічний client_id для входу через Microsoft")

49 limits: Limits 

50 load_levels: Dict[str, LoadLevelHours] = Field(..., description="Межі годин на тиждень для рівнів навантаження")

54@router.get( 

55 "/config",
